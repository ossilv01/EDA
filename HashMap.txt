template <class Key, class Val, class Hash = std::hash<Key>, 
	 class Pred = std::equal_to<Key>>

class unordered_map{
public:
   //Definicion clave valor
    using clave_valor = pair<const Key, Val>;
protected:
  
   //Dentro de esos nodos se almacenan parejas clave valor
   struct ListNode {
     clave_valor cv; 
     Link sig;
     ListNode (clave_valor const&e, Link s = nullptr) : cv(e), sig (s) {}
    };
    using Link = ListNode* //Alias
   
    static const int TAM_INICIAL = 16; //Es mejor utilizar numeros primos = Teoria matemática por detrás
    static const int MAX_CARGA = 75; //máximo porcentaje de ocupación permitido


    vector<Link> array; //vector de punteros a nodos (de listas[enlaces]) 
    int nelems;  //NO es el size del vector, ES el numero de parejas (pair)
    //Objetos funcion: Acceso rápido de tener
    Hash hash;   //Funcion de hash
    Pred pred;   //Operador de comparacion "==". Para comparar claves
}

Localización [función auxiliar]:
_____________________________________
Intro: La Búsqueda se basa en 2 pasos. Encontrar el grupo en el que estoy (lo hace hash()), y 
luego busqueda lineal del nodo que contiene la clave que estoy buscando.

Buscar nodo con clave dada. Busqueda lineal en listas
Se devuelve puntero al nodo buscado (nullptr si no se encuentra)
 y un puntero al nodo anterior (necesario para la eliminacion)


bool localizar(Key const&c, Link& ant, Link& pos) const {// pos = primero de la lista correspondiente
  ant = nullptr;
  while (pos != nullptr) {
     if (pred(c, pos->cv.fisrt)) return true //recordatorio de que pred es un objeto funcion de comparación "==" 
     else {
//incremento
      ant = pos; pos = pos->sig;
    }
  }
  return false;
}

bool erase(Key const& c) 
//misma inicio, declarando indice, nodos anterior y posicion
	int i = hash(c) % array.size();
	Link ant, pos = array[i];
//usamos funcion bool localizar
	if (localizar(c, ant, pos)){
	if (ant == nullptr) //si el nodo que justo busco es el primero de la lista
		array[i] = pos->sig;
	else
		ant->sig = pos->sig; //nueva asignacion para el siguiente del anterior
	delete pos; //se borra pos
	--nelems;
	return true;
	} else
     return false;
}

Operaciones at y []
_____________________________

El at devuelve el segundo elemento de la clave, dado el primer elemento como argumento del metodo
Val const& at(Key const& c) const{
	int i = hash(c) % array.size();
	Link ant, pos = array[i];
//usamos funcion bool localizar
	if (localizar(c, ant, pos)) return pos->cv.second;
	else throw out_of_range("No se encuentra");
}
